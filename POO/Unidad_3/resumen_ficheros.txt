# Manejo de archivos con try - except - else - finally

try:
    # Intentar abrir un archivo en modo lectura
    with open("ejemplo.txt", "r", encoding="utf-8") as archivo:
        contenido = archivo.read()

except IOError as io:
    print("Error IO: ",io)

except FileNotFoundError:
    print("Error: El archivo no existe.")

else:
    # Se ejecuta solo si NO hubo excepción
    print("Contenido del archivo:")
    print(contenido)

finally:
    # Se ejecuta siempre
    print("Programa finalizado.")



# Manejo de excepción PermissionError

try:
    # Intentar escribir en un archivo protegido (ejemplo en raíz del sistema)
    with open("/protegido.txt", "w", encoding="utf-8") as archivo:
        archivo.write("Intentando escribir en un archivo protegido...")

except PermissionError:
    print("Error: No tienes permisos para escribir en este archivo.")

else:
    print("Archivo escrito correctamente.")

finally:
    print("Programa finalizado.")
